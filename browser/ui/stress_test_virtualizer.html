Rust Backend Integration (Axum & SSE)

To stream hundreds of pages from a Rust web server directly into this virtualized frontend without buffering the entire output in memory, use Server-Sent Events (SSE) with Axum or Actix-web.

Cargo.toml 
[dependencies]
axum = { version = "0.7", features = ["macros"] }
tokio = { version = "1.0", features = ["full"] }
tokio-stream = "0.1"
futures = "0.3"
serde = { version = "1.0", features = ["derive"] }
serde_json = "1.0"
src/main.rs (High-Speed SSE Streamer) 
use axum::{
    response::sse::{Event, KeepAlive, Sse},
    routing::get,
    Router,
};
use futures::stream::Stream;
use std::{convert::Infallible, time::Duration};
use tokio_stream::StreamExt;

#[tokio::main]
async fn main() {
    let app = Router::new().route("/api/chat/stream", get(chat_stream_handler));
    let listener = tokio::net::TcpListener::bind("127.0.0.1:3000").await.unwrap();
    println!("Chat SSE Stream Server running on http://127.0.0.1:3000");
    axum::serve(listener, app).await.unwrap();
}

async fn chat_stream_handler() -> Sse<impl Stream<Item = Result<Event, Infallible>>> {
    // Generate simulated sequence across hundreds of pages
    let total_chunks = 5000; // e.g., ~150-200 standard book pages
    let stream = tokio_stream::iter(0..total_chunks).map(|i| {
        let chunk_data = serde_json::json!({
            "id": i,
            "role": if i % 2 == 0 { "assistant" } else { "user" },
            "page": (i / 25) + 1,
            "tokens": 45,
            "content": format!("Chunk #{}: Evaluated sequence token retention in KV cache.", i)
        });
        Ok(Event::default().json_data(chunk_data).unwrap())
    })
    .throttle(Duration::from_millis(30)); // Emulate LLM generation rate

    Sse::new(stream).keep_alive(KeepAlive::default())
}
